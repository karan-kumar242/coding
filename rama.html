document.addEventListener("DOMContentLoaded", () => { const videos = [ { url: "https://www.w3schools.com/html/mov_bbb.mp4", title: "Coding Short 💻" }, { url: "https://www.w3schools.com/html/movie.mp4", title: "Travel Short ✈️" }, { url: "https://www.w3schools.com/html/mov_bbb.mp4", title: "Fun Short 😄" } ]; const container = document.getElementById("shortsContainer"); videos.forEach((video) => { const wrapper = document.createElement("div"); wrapper.classList.add("videoWrapper"); wrapper.innerHTML = `
${video.title}
`; container.appendChild(wrapper); }); const allVideos = document.querySelectorAll(".video"); const observer = new IntersectionObserver( (entries) => { entries.forEach((entry) => { if (entry.isIntersecting) { entry.target.play(); } else { entry.target.pause(); } }); }, { threshold: 0.7 } ); allVideos.forEach((video) => observer.observe(video)); });
